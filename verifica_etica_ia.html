12 pt

Leggi il codice seguente. Rappresenta un sistema di raccomandazione per borse di studio universitarie. Contiene scelte di design che introducono bias. Il tuo compito è analizzarlo e riscriverlo.

# Sistema di selezione borse di studio — versione originale class Candidato: def __init__(self, nome, voto_maturita, reddito_familiare, residenza, hobby): self.nome = nome self.voto_maturita = voto_maturita self.reddito_familiare = reddito_familiare self.residenza = residenza # es. "nord", "sud", "estero" self.hobby = hobby # es. ["calcio", "musica", "coding"] class SistemaBorse: BONUS_RESIDENZA = {"nord": 10, "centro": 5, "sud": 0, "estero": -5} BONUS_HOBBY = {"coding": 8, "musica": 3, "calcio": 0, "altro": -2} def calcola_punteggio(self, candidato): score = candidato.voto_maturita # base: 60–100 score += self.BONUS_RESIDENZA.get(candidato.residenza, 0) for h in candidato.hobby: score += self.BONUS_HOBBY.get(h, -2) if candidato.reddito_familiare > 50000: score -= 20 return score def seleziona(self, candidati): ranked = sorted(candidati, key=self.calcola_punteggio, reverse=True) return ranked[:3] # restituisce i top 3
